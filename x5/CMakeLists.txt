cmake_minimum_required(VERSION 3.16)
project(insta360_x5_client VERSION 0.8.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_PYTHON "Build the Python latest/read interface" OFF)

set(CAMERASDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/CameraSDK-2.2.4-linux-x86_64"
    CACHE PATH "CameraSDK directory containing include/ and lib/libCameraSDK.so")
# Refresh derived paths when an existing build switches between SDK locations.
unset(CAMERASDK_INCLUDE_DIR CACHE)
unset(CAMERASDK_LIBRARY CACHE)
find_path(CAMERASDK_INCLUDE_DIR camera/camera.h PATHS "${CAMERASDK_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(CAMERASDK_LIBRARY CameraSDK PATHS "${CAMERASDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${CAMERASDK_ROOT}/bin/jsons/camera_conf_Insta360_X5.json")
    message(FATAL_ERROR "CAMERASDK_ROOT is missing bin/jsons/camera_conf_Insta360_X5.json")
endif()
add_library(CameraSDK SHARED IMPORTED)
set_target_properties(CameraSDK PROPERTIES IMPORTED_LOCATION "${CAMERASDK_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${CAMERASDK_INCLUDE_DIR}")
find_package(yaml-cpp REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core highgui imgproc imgcodecs)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)

add_library(x5_config src/config.cpp)
target_include_directories(x5_config PUBLIC include)
# Only enum constants are used; offline configuration checks do not call the SDK.
target_include_directories(x5_config PRIVATE "${CAMERASDK_INCLUDE_DIR}")
target_link_libraries(x5_config PUBLIC yaml-cpp::yaml-cpp)
add_library(x5_capture src/capture.cpp src/decoder.cpp)
target_include_directories(x5_capture PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(x5_capture PUBLIC opencv_core PRIVATE PkgConfig::FFMPEG Threads::Threads)
add_library(x5_preview src/preview.cpp)
target_include_directories(x5_preview PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(x5_preview PUBLIC x5_config x5_capture opencv_imgproc)
add_library(x5_session src/session.cpp src/camera.cpp src/reader.cpp)
target_link_libraries(x5_session PUBLIC x5_preview PRIVATE CameraSDK opencv_highgui Threads::Threads)
target_compile_definitions(x5_session PRIVATE X5_VERSION="${PROJECT_VERSION}")
add_executable(insta360_x5_client src/main.cpp)
target_link_libraries(insta360_x5_client PRIVATE x5_session)
target_compile_definitions(insta360_x5_client PRIVATE X5_VERSION="${PROJECT_VERSION}")
set_target_properties(insta360_x5_client PROPERTIES BUILD_RPATH "${CAMERASDK_ROOT}/lib")
# The SDK loads capability JSONs next to the executable. CREATE_LINK replaces
# an existing link so SDK overrides update headers, library and JSONs together.
file(CREATE_LINK "${CAMERASDK_ROOT}/bin/jsons" "${CMAKE_CURRENT_BINARY_DIR}/jsons" SYMBOLIC)
foreach(target x5_config x5_capture x5_preview x5_session insta360_x5_client)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

if(BUILD_PYTHON)
    find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
    set(PYBIND11_FINDPYTHON ON)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(insta360_x5 python/bindings.cpp)
    target_link_libraries(insta360_x5 PRIVATE x5_session)
    target_compile_definitions(insta360_x5 PRIVATE X5_VERSION="${PROJECT_VERSION}")
    set_target_properties(insta360_x5 PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/python"
        BUILD_RPATH "${CAMERASDK_ROOT}/lib" INSTALL_RPATH "$ORIGIN")
    install(TARGETS insta360_x5 LIBRARY DESTINATION .)
    install(FILES "${CAMERASDK_LIBRARY}" DESTINATION .)
    install(DIRECTORY "${CAMERASDK_ROOT}/bin/jsons" DESTINATION insta360_x5_data)
    install(FILES third_party/CameraSDK-2.2.4-linux-x86_64/README.md DESTINATION insta360_x5_data/CameraSDK)
    install(FILES third_party/insta360_driver/README.md third_party/insta360_driver/LICENSE
        DESTINATION insta360_x5_data/insta360_driver)
endif()
