cmake_minimum_required(VERSION 3.16)
project(go3s_bridge VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
add_library(go3s_core src/config.cpp src/wire.cpp src/decoder.cpp src/capture.cpp src/adb.cpp src/reader.cpp)
set_target_properties(go3s_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(go3s_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(go3s_core PUBLIC yaml-cpp nlohmann_json::nlohmann_json Threads::Threads PkgConfig::FFMPEG ${OpenCV_LIBS})
target_compile_options(go3s_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(insta360_go3s_client src/main.cpp)
target_link_libraries(insta360_go3s_client PRIVATE go3s_core)
option(GO3S_BUILD_PYTHON "Build the optional Python latest/read interface" OFF)
if(GO3S_BUILD_PYTHON)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  set(PYBIND11_FINDPYTHON ON)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native MODULE src/python.cpp)
  target_link_libraries(_native PRIVATE go3s_core)
  install(TARGETS _native LIBRARY DESTINATION insta360_go3s COMPONENT python)
  install(FILES python/insta360_go3s/__init__.py DESTINATION insta360_go3s COMPONENT python)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
  add_executable(go3s_unit_tests tests/unit.cpp)
  target_link_libraries(go3s_unit_tests PRIVATE go3s_core)
  add_executable(go3s_synthetic_client tests/synthetic_client.cpp)
  target_link_libraries(go3s_synthetic_client PRIVATE go3s_core)
  add_executable(go3s_low_latency_tests tests/low_latency.cpp)
  target_link_libraries(go3s_low_latency_tests PRIVATE go3s_core)
  add_test(NAME low_latency COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/low_latency.py $<TARGET_FILE:go3s_low_latency_tests>)
  add_test(NAME synthetic_unit COMMAND go3s_unit_tests)
  add_test(NAME synthetic_adb_selection COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/adb_selection.py $<TARGET_FILE:insta360_go3s_client>)
  add_test(NAME synthetic_bridge_integration COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:go3s_synthetic_client>
    $<TARGET_FILE:insta360_go3s_client> ${FFMPEG_EXECUTABLE})
  set_tests_properties(synthetic_bridge_integration PROPERTIES TIMEOUT 90)
  if(GO3S_BUILD_PYTHON)
    file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/python/insta360_go3s)
    configure_file(python/insta360_go3s/__init__.py python/insta360_go3s/__init__.py COPYONLY)
    set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/python/insta360_go3s)
    add_test(NAME python_reader COMMAND ${Python3_EXECUTABLE} -m pytest -q
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_python_reader.py)
    set_tests_properties(python_reader PROPERTIES TIMEOUT 90
      ENVIRONMENT "PYTHONPATH=${CMAKE_CURRENT_BINARY_DIR}/python")
  endif()
endif()
install(TARGETS insta360_go3s_client RUNTIME DESTINATION bin)
